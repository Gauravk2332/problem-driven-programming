Chapter 07 / 13

Keep the more flexible ending

Two length-two sequences end at 3 and 5. A future 4 can extend only the first, while 6 can extend both. For the same length the smaller ending is never worse. Keep one minimum record per length.

tails[k] = smallest possible ending of an increasing subsequence of length k+1
p = first index with tails[p] ≥ x
append if p = size; otherwise replace tails[p]

For equal lengths a smaller ending can extend to every value a larger ending can extend to. Minimum tails strictly increase: deleting the last element of a sequence gives a shorter sequence with smaller end. Thus lower_bound is valid. At position p, all previous shorter ends are <x, so length p+1 is attainable by appending x; a longer sequence ending at x would require an existing length p+1 tail <x, impossible by the search rule. Only record p changes. The number of records equals the LIS length.

Top-down and bottom-up

The O(n²) endpoint DP can be written top-down or bottom-up. The faster O(n log n) method is a different, greedy-plus-binary-search idea and has no top-down version.

Top-down (memoization)Bottom-up (tabulation)
Starts fromEach endpoint: ending_at(i)Index 0, then each index left to right
Stored byA memo keyed by endpoint iAn array dp[0…n−1], filled left to right
How it runsRecursion. A state is solved the first time it is asked for, then read from the cacheLoops. Every state is filled once, after the states it depends on
Which statesOnly the states the final answer actually needsUsually every state in the table
Extra costA call stack as deep as the longest chain of calls; very deep inputs can overflow itNo recursion; the table can often be shrunk to one or two rows
Choose it whenThe fill order is awkward, or many states are never neededThe fill order is clear and most states are needed; this is the safer default for large inputs

The answer is the best endpoint, so top-down asks every ending_at(i) and takes the maximum; the memo makes sure each endpoint is still solved only once.

Python · lis_two_ways.py

def lis_top_down(a):
    # Top-down: ending_at(i) = longest strictly increasing subsequence ending at i
    memo = {}

    def ending_at(i):
        if i not in memo:
            memo[i] = 1 + max((ending_at(j) for j in range(i) if a[j] < a[i]), default=0)
        return memo[i]

    return max((ending_at(i) for i in range(len(a))), default=0)


def lis_bottom_up(a):
    # Bottom-up: fill dp[0], dp[1], ... left to right
    dp = [1] * len(a)
    for i in range(len(a)):
        for j in range(i):
            if a[j] < a[i]:
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp, default=0)


scores = [10, 9, 2, 5, 3, 7, 101, 18]
print(lis_top_down(scores), lis_bottom_up(scores))  # 4 4
print(lis_top_down([]), lis_bottom_up([]))          # 0 0

#include <algorithm>
#include <iostream>
#include <vector>
using namespace std;

vector<int> a, memo;  // memo[i] = 0 means "not computed yet" (real answers are >= 1)

// Top-down: endingAt(i) = longest strictly increasing subsequence ending at i
int endingAt(int i) {
    if (memo[i]) return memo[i];
    int best = 1;
    for (int j = 0; j < i; j++)
        if (a[j] < a[i]) best = max(best, 1 + endingAt(j));
    return memo[i] = best;
}

int lisTopDown() {
    memo.assign(a.size(), 0);
    int ans = 0;
    for (int i = 0; i < (int)a.size(); i++) ans = max(ans, endingAt(i));
    return ans;
}

// Bottom-up: fill dp[0], dp[1], ... left to right
int lisBottomUp() {
    vector<int> dp(a.size(), 1);
    int ans = 0;
    for (int i = 0; i < (int)a.size(); i++) {
        for (int j = 0; j < i; j++)
            if (a[j] < a[i]) dp[i] = max(dp[i], dp[j] + 1);
        ans = max(ans, dp[i]);
    }
    return ans;
}

int main() {
    a = {10, 9, 2, 5, 3, 7, 101, 18};
    cout << lisTopDown() << " " << lisBottomUp() << "\n";  // 4 4
    a = {};
    cout << lisTopDown() << " " << lisBottomUp() << "\n";  // 0 0
}

Java · LISTwoWays.java

public class LISTwoWays {
    static int[] a, memo;  // memo[i] = 0 means "not computed yet" (real answers are >= 1)

    // Top-down: endingAt(i) = longest strictly increasing subsequence ending at i
    static int endingAt(int i) {
        if (memo[i] != 0) return memo[i];
        int best = 1;
        for (int j = 0; j < i; j++)
            if (a[j] < a[i]) best = Math.max(best, 1 + endingAt(j));
        return memo[i] = best;
    }

    static int lisTopDown() {
        memo = new int[a.length];
        int ans = 0;
        for (int i = 0; i < a.length; i++) ans = Math.max(ans, endingAt(i));
        return ans;
    }

    // Bottom-up: fill dp[0], dp[1], ... left to right
    static int lisBottomUp() {
        int[] dp = new int[a.length];
        int ans = 0;
        for (int i = 0; i < a.length; i++) {
            dp[i] = 1;
            for (int j = 0; j < i; j++)
                if (a[j] < a[i]) dp[i] = Math.max(dp[i], dp[j] + 1);
            ans = Math.max(ans, dp[i]);
        }
        return ans;
    }

    public static void main(String[] args) {
        a = new int[]{10, 9, 2, 5, 3, 7, 101, 18};
        System.out.println(lisTopDown() + " " + lisBottomUp());  // 4 4
        a = new int[]{};
        System.out.println(lisTopDown() + " " + lisBottomUp());  // 0 0
    }
}

from bisect import bisect_left

def quadratic(a):
    d=[1]*len(a); p=[-1]*len(a)
    for i in range(len(a)):
        for j in range(i):
            if a[j]<a[i] and d[j]+1>d[i]:
                d[i]=d[j]+1; p[i]=j
    i=max(range(len(a)),key=lambda j:d[j]) if a else -1
    path=[]
    while i!=-1:
        path.append(i); i=p[i]
    return path[::-1]

def fast(a):
    tails=[]; ends=[]; p=[-1]*len(a)
    for i,x in enumerate(a):
        k=bisect_left(tails,x)
        if k: p[i]=ends[k-1]
        if k==len(tails):
            tails.append(x); ends.append(i)
        else:
            tails[k]=x; ends[k]=i
    i=ends[-1] if ends else -1
    path=[]
    while i!=-1:
        path.append(i); i=p[i]
    return path[::-1]

if __name__=="__main__":
    a=[10,9,2,5,3,7,101,18]
    for f in [quadratic,fast]:
        ids=f(a)
        print(len(ids),[a[i] for i in ids])

vector<int> quadratic(const vector<int>& a){
    int n=a.size(),best=-1;
    vector<int> d(n,1),p(n,-1);
    for(int i=0;i<n;i++){

Java · LIS.java

import java.util.*;
public class LIS {
    static List<Integer> restore(int i,int[] p){
        List<Integer> out=new ArrayList<>();
        while(i!=-1){out.add(i);i=p[i];}
        Collections.reverse(out);return out;
    }
    static List<Integer> quadratic(int[] a){
        int n=a.length,best=-1;
        int[] d=new int[n],p=new int[n];
        Arrays.fill(d,1);Arrays.fill(p,-1);
        for(int i=0;i<n;i++){
            for(int j=0;j<i;j++)
                if(a[j]<a[i] && d[j]+1>d[i]){
                    d[i]=d[j]+1;p[i]=j;
                }
            if(best<0 || d[i]>d[best])best=i;
        }
        return restore(best,p);
    }
    static List<Integer> fast(int[] a){
        int n=a.length,len=0;
        int[] t=new int[n],ends=new int[n],p=new int[n];
        Arrays.fill(p,-1);
        for(int i=0;i<n;i++){
            int l=0,r=len;
            while(l<r){
                int mid=l+(r-l)/2;
                if(t[mid]<a[i])l=mid+1;else r=mid;
            }
            if(l>0)p[i]=ends[l-1];
            t[l]=a[i];ends[l]=i;
            if(l==len)len++;
        }
        return restore(len==0?-1:ends[len-1],p);
    }
    public static void main(String[] args){
        int[] a={10,9,2,5,3,7,101,18};
        for(List<Integer> ids:Arrays.asList(quadratic(a),fast(a))){
            List<Integer> v=new ArrayList<>();
            for(int i:ids)v.add(a[i]);
            System.out.println(ids.size()+": "+v);
        }
    }
}

C1 · Derive, compute, reconstruct quadratic LIS. 10 marks

Reveal model answer
dp[i] = 1 + max(dp[j] : j<i and a[j]<a[i])
Empty predecessor maximum = 0
Answer = max(dp); empty input → 0

Every increasing subsequence ending at i is either the singleton a[i] or has a penultimate index j<i with a[j]<a[i]. Its prefix is bounded by dp[j]; conversely each such optimal prefix can append a[i]. Taking the maximum gives exactly the best endpoint length. Fill left to right since dependencies are earlier indices.
