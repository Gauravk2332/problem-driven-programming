Chapter 04 / 13

Fill the edges before the interior

Empty source to abc means inserting a, b, c. abc to empty means deleting them. So zero row grows 0,1,2,3,... across; zero column grows down. These are known answers, not padding.

Find what to change

The number 3 tells us the distance, but not which characters to edit. Start at the final cost and find any neighbour whose cost plus its edit equals the current cost. Travel backwards, then reverse the chosen alignment columns.

Chapter 10 / 13

Save memory

Think of the table as rows of notes. While writing the current row you need the previous row and your latest note to the left. Older rows can be discarded if you only want the final number.

Chapter 11 / 13

Translate into code and practise

The recursive implementation mirrors the three cases. The table calculates them in dependency order. The rolling version retains two rows, while alignment reconstruction uses the full table.

Top-down and bottom-up

Edit distance can be computed either way; the programs below include both.

Top-down (memoization)Bottom-up (tabulation)
Starts fromThe full strings: f(m, n)Row 0 and column 0 (empty prefixes)
Stored byA cache keyed by (i, j)A table D[i][j], filled row by row, left to right
How it runsRecursion. A state is solved the first time it is asked for, then read from the cacheLoops. Every state is filled once, after the states it depends on
Which statesOnly the states the final answer actually needsUsually every state in the table
Extra costA call stack as deep as the longest chain of calls; very deep inputs can overflow itNo recursion; the table can often be shrunk to one or two rows
Choose it whenThe fill order is awkward, or many states are never neededThe fill order is clear and most states are needed; this is the safer default for large inputs

Recursion depth can reach m + n, so for long strings prefer bottom-up. In the programs below, memoized is top-down, table is bottom-up, and rolling is bottom-up with two rows.

Python · edit_distance.py

from functools import lru_cache

def memoized(a,b):
    @lru_cache(None)
    def f(i,j):
        if i==0: return j
        if j==0: return i
        return min(f(i-1,j)+1,f(i,j-1)+1,
                   f(i-1,j-1)+(a[i-1]!=b[j-1]))
    return f(len(a),len(b))

def table(a,b):
    d=[[0]*(len(b)+1) for _ in range(len(a)+1)]
    for i in range(len(a)+1): d[i][0]=i
    for j in range(len(b)+1): d[0][j]=j
    for i in range(1,len(a)+1):
        for j in range(1,len(b)+1):
            d[i][j]=min(d[i-1][j]+1,d[i][j-1]+1,
                        d[i-1][j-1]+(a[i-1]!=b[j-1]))
    return d

def rolling(a,b):
    if len(b)>len(a): a,b=b,a
    prev=list(range(len(b)+1))
    for i,x in enumerate(a,1):
        cur=[i]+[0]*len(b)
        for j,y in enumerate(b,1):
            cur[j]=min(prev[j]+1,cur[j-1]+1,
                       prev[j-1]+(x!=y))
        prev=cur
    return prev[-1]

def alignment(a,b):
    d=table(a,b); i,j=len(a),len(b); out=[]
    while i or j:
        if i and j and d[i][j]==d[i-1][j-1]+(a[i-1]!=b[j-1]):
            out.append((a[i-1],b[j-1])); i-=1; j-=1
        elif i and d[i][j]==d[i-1][j]+1:
            out.append((a[i-1],'')); i-=1
        else:
            out.append(('',b[j-1])); j-=1
    return out[::-1]

if __name__=='__main__':
    a,b='horse','ros'
    print(memoized(a,b),table(a,b)[-1][-1],rolling(a,b))
    print(alignment(a,b))

#include <iostream>
#include <vector>
#include <string>
#include <algorithm>
#include <functional>
#include <utility>
using namespace std;
int memoized(const string& a,const string& b){
    vector<vector<int>> d(a.size()+1,vector<int>(b.size()+1,-1));
    function<int(int,int)> f=[&](int i,int j){
        if(!i)return j;
        if(!j)return i;
        int& v=d[i][j];if(v!=-1)return v;
        return v=min({f(i-1,j)+1,f(i,j-1)+1,
                     f(i-1,j-1)+(a[i-1]!=b[j-1])});
    };
    return f(a.size(),b.size());
}
vector<vector<int>> table(const string& a,const string& b){
    int m=a.size(),n=b.size();
    vector<vector<int>> d(m+1,vector<int>(n+1));
    for(int i=0;i<=m;i++)d[i][0]=i;
    for(int j=0;j<=n;j++)d[0][j]=j;
    for(int i=1;i<=m;i++)for(int j=1;j<=n;j++)
        d[i][j]=min({d[i-1][j]+1,d[i][j-1]+1,
                     d[i-1][j-1]+(a[i-1]!=b[j-1])});
    return d;
}
int rolling(const string& a,const string& b){
    if(b.size()>a.size())return rolling(b,a);
    int m=a.size(),n=b.size();
    vector<int> prev(n+1),cur(n+1);
    for(int j=0;j<=n;j++)prev[j]=j;
    for(int i=1;i<=m;i++){
        cur[0]=i;
        for(int j=1;j<=n;j++)
            cur[j]=min({prev[j]+1,cur[j-1]+1,
                        prev[j-1]+(a[i-1]!=b[j-1])});
        swap(prev,cur);
    }
    return prev[n];
}
vector<pair<string,string>> alignment(const string& a,const string& b){
    auto d=table(a,b);int i=a.size(),j=b.size();
    vector<pair<string,string>> out;
    while(i||j){
        if(i&&j&&d[i][j]==d[i-1][j-1]+(a[i-1]!=b[j-1])){
            out.push_back({a.substr(i-1,1),b.substr(j-1,1)});--i;--j;
        }else if(i&&d[i][j]==d[i-1][j]+1){
            out.push_back({a.substr(i-1,1),""});--i;
        }else{out.push_back({"",b.substr(j-1,1)});--j;}
    }
    reverse(out.begin(),out.end());return out;
}
int main(){
    string a="horse",b="ros";
    cout<<memoized(a,b)<<" "<<table(a,b).back().back()<<" "<<rolling(a,b)<<"\n";
    for(auto x:alignment(a,b))
        cout<<"("<<x.first<<","<<x.second<<") ";
    cout<<"\n";
}

import java.util.*;
public class EditDistance {
    static int min(int a,int b,int c){return Math.min(a,Math.min(b,c));}
    static int f(String a,String b,int i,int j,int[][] d){
        if(i==0)return j;
        if(j==0)return i;
        if(d[i][j]!=-1)return d[i][j];
        return d[i][j]=min(f(a,b,i-1,j,d)+1,f(a,b,i,j-1,d)+1,
            f(a,b,i-1,j-1,d)+(a.charAt(i-1)==b.charAt(j-1)?0:1));
    }
    static int memoized(String a,String b){
        int[][] d=new int[a.length()+1][b.length()+1];
        for(int[] r:d)Arrays.fill(r,-1);
        return f(a,b,a.length(),b.length(),d);
    }
    static int[][] table(String a,String b){
        int m=a.length(),n=b.length();int[][] d=new int[m+1][n+1];
        for(int i=0;i<=m;i++)d[i][0]=i;
        for(int j=0;j<=n;j++)d[0][j]=j;
        for(int i=1;i<=m;i++)for(int j=1;j<=n;j++)
            d[i][j]=min(d[i-1][j]+1,d[i][j-1]+1,
                d[i-1][j-1]+(a.charAt(i-1)==b.charAt(j-1)?0:1));
        return d;
    }
    static int rolling(String a,String b){
        if(b.length()>a.length())return rolling(b,a);
        int n=b.length();int[] prev=new int[n+1],cur=new int[n+1];
        for(int j=0;j<=n;j++)prev[j]=j;
        for(int i=1;i<=a.length();i++){
            cur[0]=i;
            for(int j=1;j<=n;j++)cur[j]=min(prev[j]+1,cur[j-1]+1,
                prev[j-1]+(a.charAt(i-1)==b.charAt(j-1)?0:1));
            int[] tmp=prev;prev=cur;cur=tmp;
        }
        return prev[n];
    }
    static List<String> alignment(String a,String b){
        int[][] d=table(a,b);int i=a.length(),j=b.length();
        List<String> out=new ArrayList<>();
        while(i>0||j>0){
            if(i>0&&j>0&&d[i][j]==d[i-1][j-1]+(a.charAt(i-1)==b.charAt(j-1)?0:1)){
                out.add("("+a.charAt(i-1)+","+b.charAt(j-1)+")");i--;j--;
            }else if(i>0&&d[i][j]==d[i-1][j]+1){
                out.add("("+a.charAt(i-1)+",)");i--;
            }else{out.add("(,"+b.charAt(j-1)+")");j--;}
        }
        Collections.reverse(out);return out;
    }
    public static void main(String[] args){
        String a="horse",b="ros";
        System.out.println(memoized(a,b)+" "+table(a,b)[a.length()][b.length()]+" "+rolling(a,b));
        System.out.println(alignment(a,b));
    }
}

Identical strings distance 0; empty to length n distance n; |m−n| ≤ distance ≤ max(m,n), since every edit changes length by at most one and one can replace the shared-length part then insert/delete the rest. Unit-cost Levenshtein distance is symmetric; swapping strings lets the shorter suffix dimension be columns for memory. For weighted asymmetric edits, swapping may change the problem. Transposition is not a single operation here: ab→ba costs 2. Unicode graphemes may need explicit tokenization; these examples use ASCII. Hamming distance only compares equal-length aligned positions. With insert/delete only, distance = m+n−2L where L is LCS length; that formula does not generally hold when replacement costs 1.

An optimal unit-cost edit script has a normal-form alignment: each source character either pairs with one target character in order or is deleted; unpaired target characters are inserted. Repeated changes to the same character and inserted-then-deleted characters can be removed without increasing cost. Thus a final alignment column is either a deletion, insertion or paired characters (match/replacement). Removing that column leaves an alignment of the corresponding smaller prefixes. If that alignment were not optimal we could replace it with a cheaper one. Conversely appending each legal column to an optimal smaller alignment produces a feasible script. The minimum of the three costs is therefore exact. The empty-prefix costs follow because every remaining character needs one insertion or deletion. All dependencies decrease i+j.
